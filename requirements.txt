# Upper bounds are deliberate. Streamlit Cloud installs the latest matching
# version at build time, so an unbounded ">=" means a major release from any of
# these could break the deployment on a date nobody chose. The Anthropic SDK in
# particular has a 1.x line with breaking changes (httpx2, removed parameters),
# and this project must keep working through a fixed submission deadline.
streamlit>=1.40,<2
anthropic>=0.40,<1
pandas>=2.0,<4
altair>=5,<7             # charts; Streamlit ships it, but app.py imports it directly
fastapi>=0.115,<1        # web API for the React app (api/server.py)
uvicorn>=0.30,<1         # serves the web API

# Optional: only needed to scan victims from other vendors. The app runs fine
# without them and shows those models as unavailable.
# openai>=1.40,<2          # GPT-4o mini, GPT-4.1 mini, and Groq/Together via base_url
# google-genai>=1.0,<2     # Gemini 2.0 Flash

# Native review sheets: building them (review_sheet.py --xlsx) and reading what
# reviewers send back (review_ingest.py, whose self-test runs in CI). Never
# imported by the app or the API.
openpyxl>=3.1,<4
